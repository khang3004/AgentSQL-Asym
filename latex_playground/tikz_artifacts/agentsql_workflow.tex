\begin{tikzpicture}[
        node distance=1.2cm and 2.5cm,
        font=\sffamily\footnotesize,
        >=Stealth,
        % --- Styles ---
        phase/.style={
                rectangle, minimum width=5.2cm, minimum height=1.1cm,
                text centered, rounded corners=4pt, line width=0.7pt,
                blur shadow={shadow blur steps=6, shadow xshift=0.5pt, shadow yshift=-1pt}
            },
        io/.style={
                trapezium, trapezium left angle=70, trapezium right angle=110,
                minimum width=3.5cm, minimum height=0.9cm,
                text centered, draw=io_border, fill=io_fill,
                line width=0.8pt, blur shadow={shadow blur steps=4}
            },
        decision/.style={
                diamond, aspect=2.5, minimum width=3.2cm,
                inner sep=1pt, text centered, draw=sandbox_border, fill=sandbox_fill,
                line width=0.8pt, blur shadow={shadow blur steps=4}
            },
        arrow/.style={thick, ->, line width=0.8pt, draw=black!55},
        feedback/.style={thick, ->, dashed, line width=0.8pt, draw=red!55},
        label/.style={font=\sffamily\scriptsize, text=black!60}
    ]

    % ── Colour Palette ──────────────────────────────────────────────────
    \definecolor{chess_fill}{RGB}{227, 242, 253}
    \definecolor{chess_border}{RGB}{33, 150, 243}
    \definecolor{mci_fill}{RGB}{243, 229, 245}
    \definecolor{mci_border}{RGB}{156, 39, 176}
    \definecolor{asm_fill}{RGB}{232, 245, 233}
    \definecolor{asm_border}{RGB}{76, 175, 80}
    \definecolor{gen_fill}{RGB}{232, 245, 233}
    \definecolor{gen_border}{RGB}{76, 175, 80}
    \definecolor{reflect_fill}{RGB}{255, 253, 231}
    \definecolor{reflect_border}{RGB}{255, 179, 0}
    \definecolor{critic_fill}{RGB}{255, 235, 238}
    \definecolor{critic_border}{RGB}{229, 57, 53}
    \definecolor{sandbox_fill}{RGB}{236, 239, 241}
    \definecolor{sandbox_border}{RGB}{96, 125, 139}
    \definecolor{io_fill}{RGB}{255, 248, 225}
    \definecolor{io_border}{RGB}{255, 160, 0}

    % ── LangGraph Asymmetric State Machine Nodes ────────────────────────

    % Input (row 0)
    \node (input) [io] {\textbf{NL Question} $Q$ and Schema $\mathcal{S}$};

    % Node 1: Adaptive Router Node (row 1)
    \node (router) [phase, fill=chess_fill, draw=chess_border, below=1.3cm of input]
    {\textbf{Node 1: Adaptive Router} \\[4pt] \scriptsize \texttt{openai/gpt-oss-20b} via Groq};

    % Node 2: Schema Explorer Node (row 2) - only for COMPLEX
    \node (explorer) [phase, fill=mci_fill, draw=mci_border, below=1.2cm of router]
    {\textbf{Node 2: Schema Explorer} \\[4pt] \scriptsize CHESS Semantic Pruning + MCI Card/Range Profiling};

    % Node 3: SQL Generator Node (row 3)
    \node (gen) [phase, fill=gen_fill, draw=gen_border, below=1.4cm of explorer]
    {\textbf{Node 3: SQL Generator} \\[4pt] \scriptsize Dual Candidate (DDL vs Markdown) + Sandbox Selection};

    % Node 4: Resilient SQL Validator Node (row 4)
    \node (validator) [decision, below=1.1cm of gen]
    {\scriptsize Node 4: Resilient \\[2pt] \scriptsize SQL Validator};

    % Node 5: Output Aligner Node (row 5)
    \node (aligner) [phase, fill=reflect_fill, draw=reflect_border, below=1.1cm of validator]
    {\textbf{Node 5: Output Aligner} \\[4pt] \scriptsize Column name \& standard case alignment};

    % Output (row 6)
    \node (output) [io, below=1.1cm of aligner] {\textbf{Final SQL} $\hat{y}$};

    % Node 6: Targeted Resilient Corrector Node (right of validator)
    \node (corrector) [phase, fill=critic_fill, draw=critic_border, right=3.2cm of validator]
    {\textbf{Node 6: Targeted Corrector} \\[4pt] \scriptsize \texttt{openai/gpt-oss-120b} via Groq};

    % ── ARROWS & ROUTING ────────────────────────────────────────────────

    \draw [arrow] (input)   -- node[label, right] {} (router);
    
    % Router branch 1: COMPLEX (goes straight through Schema Explorer)
    \draw [arrow] (router)   -- node[label, right] {\textsc{COMPLEX}} (explorer);
    \draw [arrow] (explorer) -- node[label, right] {$\mathcal{S}'_{\mathrm{enriched}}$} (gen);
    
    % Router branch 2: SIMPLE (bypasses Schema Explorer on the right/left side)
    % We draw a curved arrow on the left to bypass Node 2 cleanly.
    \draw [arrow] (router.west) to [out=185, in=175, looseness=1.2] 
        node[label, left, midway, xshift=-2pt] {\textsc{SIMPLE} (bypass)} 
        (gen.west);

    \draw [arrow] (gen)     -- node[label, right] {$y_0$} (validator);
    
    % PASS path from validator to aligner
    \draw [arrow]
    ([xshift=-8pt]validator.south) --
    node[label, left] {\textsc{SUCCESS}}
    ([xshift=-8pt]aligner.north);

    \draw [arrow] (aligner) -- (output);

    % Loop: validator -> corrector on execution failure
    \draw [feedback] (validator.east) -- 
        node[label, above] {\scriptsize Syntax/Empty/Null} 
        (corrector.west);

    % Loop feedback: corrector -> validator's south input
    \draw [feedback] (corrector.south) |- ++(0,-0.8) -|
        node[label, below, near start] {\scriptsize Corrected SQL $y_{i+1}$ (Max 3)}
        ([xshift=8pt]validator.south);

    \begin{scope}[on background layer]

        % Bounding box for entire state machine
        \node[draw=chess_border!60, dashed, line width=1.0pt,
            fill=chess_fill!10, inner sep=12pt,
            rounded corners=10pt,
            fit=(router)(explorer)(gen)(validator)(corrector)(aligner)] (state_machine_box) {};

        % Bounding box label rotated on the side
        \node[anchor=center,
            rotate=90,
            font=\sffamily\bfseries\small,
            text=chess_border!85,
            fill=white, inner xsep=4pt, inner ysep=2pt]

        {AGENTSQL-ASYM LANGGRAPH STATE MACHINE};

    \end{scope}
